# Release image (no debug account): the first boot setup, then a login
# with the user it created and sudo.
wait "Hostname [nexus]: " 60
type "testbox\n"
wait "New root password: "
type "rootpw\n"
wait "Retype new root password: "
type "rootpw\n"
wait "Create a user (empty to skip): "
type "alice\n"
wait "Password for alice: "
type "alicepw\n"
wait "Retype password for alice: "
type "alicepw\n"
wait "Allow alice to use sudo [Y/n]: "
type "\n"
wait "Login: "
type "alice\n"
wait "Password: "
type "alicepw\n"
wait "alice@testbox:"
type "sudo id\n"
wait "[sudo] password for alice: "
type "alicepw\n"
wait "uid=0(root)"
screen firstboot
